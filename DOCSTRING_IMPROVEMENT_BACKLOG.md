```python
def function_name(param1: Type1, param2: Type2) -> ReturnType:
    """Brief one-line summary (imperative).
    
    Extended description explaining what the endpoint does, why it matters,
    and any important context about the operation or return value.
    
    Args:
        param1: Description of param1 and valid value ranges.
        param2: Description of param2, default behavior if applicable.
    
    Returns:
        Description of return value structure, shape, and content.
        Example: dict[str, Any] with keys 'success', 'data', 'message'.
    
    Raises:
        NotFoundError: Raised when resource_id does not exist.
        AuthenticationError: Raised when credentials are invalid.
        ValidationError: Raised when required fields are missing.
    
    Note:
        Any SDK-specific behavior (e.g., dry_run=True support, RequestParameters).
    
    Example:
        >>> response = retrieve_policy(policy_id="POL-123")
        >>> print(response['data']['policy_number'])
    """
    ...
```
